package __PKG__;

import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.content.Context;
import android.content.Intent;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.net.Uri;
import android.os.Build;
import androidx.annotation.NonNull;
import androidx.core.app.NotificationCompat;
import com.google.firebase.messaging.RemoteMessage;
import io.capawesome.capacitorjs.plugins.firebase.messaging.MessagingService;
import java.io.InputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.Map;

public class SctcMessagingService extends MessagingService {
    private static final String CH = "sctc_hub";

    @Override
    public void onMessageReceived(@NonNull RemoteMessage message) {
        super.onMessageReceived(message);
        Map<String, String> d = message.getData();
        if (d == null || d.get("title") == null) return;
        try {
            show(d, true);
        } catch (Throwable t) {
            try { show(d, false); } catch (Throwable ignored) { }
        }
    }

    private void show(Map<String, String> d, boolean rich) {
        Context c = getApplicationContext();
        NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
        if (Build.VERSION.SDK_INT >= 26 && nm.getNotificationChannel(CH) == null) {
            NotificationChannel ch = new NotificationChannel(CH, "Announcements", NotificationManager.IMPORTANCE_HIGH);
            ch.enableVibration(true);
            nm.createNotificationChannel(ch);
        }
        String title = d.get("title");
        String body = d.get("body") == null ? "" : d.get("body");
        String id = d.get("id") == null ? "" : d.get("id");
        String sub = d.get("sub");
        int small = c.getResources().getIdentifier("ic_stat_name", "drawable", c.getPackageName());
        int logo = c.getResources().getIdentifier("ic_logo", "drawable", c.getPackageName());

        Intent open = c.getPackageManager().getLaunchIntentForPackage(c.getPackageName());
        open.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TOP | Intent.FLAG_ACTIVITY_SINGLE_TOP);
        open.setAction(Intent.ACTION_VIEW);
        open.setData(Uri.parse("sctc://post/" + id));
        open.putExtra("google.message_id", "sctc-" + id);
        open.putExtra("id", id);
        int nid = (id.isEmpty() ? String.valueOf(System.nanoTime()) : id).hashCode();
        PendingIntent pi = PendingIntent.getActivity(c, nid, open, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);

        NotificationCompat.Builder b = new NotificationCompat.Builder(c, CH)
                .setSmallIcon(small != 0 ? small : android.R.drawable.ic_dialog_info)
                .setContentTitle(title)
                .setContentText(body)
                .setColor(0xFF0B6B2B)
                .setAutoCancel(true)
                .setContentIntent(pi)
                .setPriority(NotificationCompat.PRIORITY_HIGH)
                .setDefaults(NotificationCompat.DEFAULT_ALL)
                .setCategory(NotificationCompat.CATEGORY_MESSAGE);
        if (sub != null && !sub.isEmpty()) b.setSubText(sub);

        if (rich) {
            if (logo != 0) {
                Bitmap lb = BitmapFactory.decodeResource(c.getResources(), logo);
                if (lb != null) b.setLargeIcon(lb);
            }
            Bitmap pic = download(d.get("image"));
            if (pic != null) {
                b.setStyle(new NotificationCompat.BigPictureStyle()
                        .bigPicture(pic)
                        .bigLargeIcon((Bitmap) null)
                        .setBigContentTitle(title)
                        .setSummaryText(body));
            } else {
                b.setStyle(new NotificationCompat.BigTextStyle().bigText(body));
            }
        }
        nm.notify(nid, b.build());
    }

    private Bitmap download(String u) {
        if (u == null || u.isEmpty()) return null;
        HttpURLConnection con = null;
        try {
            con = (HttpURLConnection) new URL(u).openConnection();
            con.setConnectTimeout(8000);
            con.setReadTimeout(8000);
            InputStream in = con.getInputStream();
            try { return BitmapFactory.decodeStream(in); } finally { in.close(); }
        } catch (Throwable t) {
            return null;
        } finally {
            if (con != null) con.disconnect();
        }
    }
}
